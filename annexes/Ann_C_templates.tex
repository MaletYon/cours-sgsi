\section*{Plan d'une politique de sécurité (PSSI)}
\begin{enumerate}
    \item Introduction et contexte (objectifs, portée, références).
    \item Gouvernance (organisation, rôles, comité).
    \item Classification et manipulation de l'information.
    \item Contrôle d'accès.
    \item Sécurité des opérations (sauvegarde, journalisation, incidents).
    \item Conformité (audits, revue, sanctions).
\end{enumerate}
